Step 'Installing native runtimes when missing'
# n8n native modules ship prebuilt binaries for Node LTS only, so n8n gets a private portable Node 24.
$Node24 = Join-Path $Runtime 'node24'
if (-not (Test-Path (Join-Path $Node24 'node.exe'))) {
  $version = (Invoke-RestMethod https://nodejs.org/dist/index.json | Where-Object { $_.version -like 'v24.*' -and $_.lts } | Select-Object -First 1).version
  $zip = Join-Path $Runtime 'node.zip'
  Invoke-WebRequest -UseBasicParsing -OutFile $zip -Uri "https://nodejs.org/dist/$version/node-$version-win-x64.zip"
  Expand-Archive $zip -DestinationPath $Runtime -Force
  Remove-Item $zip
  Rename-Item (Join-Path $Runtime "node-$version-win-x64") 'node24'
}
$env:PATH = "$Node24;$env:PATH"
